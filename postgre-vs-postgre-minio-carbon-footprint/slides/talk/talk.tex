\section{Motivation}
\sectionframe{Motivation}

\begin{frame}{A carbon-aware choice}
  \vspace{2pt}\footnotesize
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item Smart buildings store IoT (Internet of Things) camera images with timestamps; the data tier runs \textbf{24/7}, so its energy is operational carbon.
    \item Two common ways to store the image on PostgreSQL/TimescaleDB:
      \begin{itemize}\footnotesize
        \item \textbf{PG} --- image \emph{inline} in a \texttt{BYTEA} column (via TOAST, The Oversized-Attribute Storage Technique).
        \item \textbf{PM} --- image \emph{externalised} to MinIO object storage; PostgreSQL keeps only metadata $+$ a key.
      \end{itemize}
    \item Time-series benchmarks use tiny numeric payloads and report no energy.
  \end{itemize}
  \vspace{2pt}
  \begin{rqbox}\footnotesize
    What are the performance and carbon trade-offs of inline \textbf{PG} vs.\ externalised \textbf{PM} storage across image resolutions?
  \end{rqbox}
  \takeaway{Inline simplicity vs.\ externalised object transport --- which is greener, and when?}
\end{frame}

\section{Discussion \& Conclusion}
\sectionframe{Discussion \& Conclusion}

\begin{frame}{Why the crossovers}
  \vspace{2pt}\footnotesize
  \begin{columns}[t]
    \column{0.5\textwidth}\textbf{Mechanism}
    \begin{itemize}\setlength{\itemsep}{2pt}
      \item \textbf{PG wins small writes:} batched inserts avoid a per-object request; PM pays a fixed MinIO PUT.
      \item \textbf{PM wins large payloads:} PostgreSQL still logs big \texttt{BYTEA} through its Write-Ahead Log (WAL); MinIO streams the object directly.
      \item PM keeps storage at $\approx$1.00$\times$ (vs.\ PG's 1.04$\times$).
    \end{itemize}
    \column{0.5\textwidth}\textbf{Guidelines}
    \begin{itemize}\setlength{\itemsep}{2pt}
      \item 360p--1080p ingest-heavy: \best{PG} (2--8$\times$ faster writes).
      \item 1440p mixed: it depends --- PM cuts reads $3.5\times$ and carbon.
      \item 4K--5K / read-heavy: \best{PM} ($\sim$6--8$\times$ faster reads, $\approx$half the carbon).
      \item Operational simplicity: PG $=$ one transactional boundary.
    \end{itemize}
  \end{columns}
  \takeaway{Crossover points, not a universal winner.}
\end{frame}

\begin{frame}{Conclusion}
  \vspace{2pt}\footnotesize
  \begin{itemize}\setlength{\itemsep}{3pt}
    \item \textbf{RQ answer:} whole-phase, PM emits \best{14{,}483} vs.\ PG 19{,}825~mg~CO\textsubscript{2}eq --- a \textbf{26.9\% reduction}.
    \item Crossovers: PG writes faster $\le$1440p; PM reads faster from 720p and is greener from 1440p.
    \item PM keeps $\approx$1.00$\times$ storage amplification and scales its Write-Ahead Log with metadata, not payload.
    \item \textbf{Threats:} single host; per-profile carbon is a constant-power-rate estimate; 2{,}000 rows.
    \item \textbf{Future work:} multi-node / higher concurrency; PostgreSQL Large Objects (LO) as a third architecture.
  \end{itemize}
  \takeaway{Inline for small ingest; externalised object storage for large, read-heavy, carbon-sensitive tiers.}
\end{frame}
